    public sealed class DirectedRockSource:ScriptableObject
    {
        [Serializable] public struct Edit { public string intent; public Vector3 point; public float radius; public Vector3 offset; }
        public Mesh source;
        public Mesh controlPalette;
        public Mesh result;
        public Vector3 center=new Vector3(-3.6f,-2.12f,-10.35f);
        public Vector3 size=new Vector3(5.8f,3.7f,1.9f);
        public bool fracturedTopFacesCamera;
        public Edit[] edits;

        public void Apply()
        {
            if(source==null || controlPalette==null || result==null || edits==null || edits.Any(e=>e.radius<=.001f || float.IsNaN(e.radius) || float.IsInfinity(e.radius)))throw new InvalidOperationException("Every sculpt influence needs a finite radius >0.001; refusing to alter mesh");
            var b=source.bounds;var input=source.vertices;var vertices=new Vector3[input.Length];
            var palettePoints=controlPalette.vertices;var paletteColors=controlPalette.colors;var colors=new Color[input.Length];
            for(var i=0;i<input.Length;i++)
            {
                var p=input[i]-b.center;p=new Vector3(p.x/b.extents.x,p.y/b.extents.y,p.z/b.extents.z);
                if(fracturedTopFacesCamera)p=new Vector3(p.x,p.z,-p.y);
                var original=p;
                foreach(var edit in edits)
                {
                    var distance=(original-edit.point).sqrMagnitude;var w=Mathf.Exp(-distance/(edit.radius*edit.radius));p+=edit.offset*w;
                }
                // Reserve the original ledge footprint. Cavity edits change the source surface, not topology.
                p=new Vector3(Mathf.Clamp(p.x,-1,1),Mathf.Clamp(p.y,-1,1),Mathf.Clamp(p.z,-1,1));
                vertices[i]=center+Vector3.Scale(p,size*.5f);
                var nearest=0;var best=float.MaxValue;
                for(var j=0;j<palettePoints.Length;j++){var d=(vertices[i]-palettePoints[j]).sqrMagnitude;if(d<best){best=d;nearest=j;}}
                colors[i]=paletteColors[nearest];
            }
            result.Clear();result.name="HeroLedgeCliff";result.vertices=vertices;result.triangles=source.triangles;result.colors=colors;
            result.uv=vertices.Select(p=>new Vector2(p.x*.34f,p.z*.34f)).ToArray();result.RecalculateNormals();result.RecalculateBounds();
            EditorUtility.SetDirty(result);EditorUtility.SetDirty(this);
        }
    }
